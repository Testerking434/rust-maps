================================================================
SCHRITT 1 — .env erstellen (NUR 3 Zeilen nötig!)
================================================================

cat > /var/www/genselfcore/.env << 'EOF'
PORT=8001
NODE_ENV=production
JWT_SECRET=Selfcore2024SuperGeheimXyzAbcDef987!
EOF

echo "✅ .env erstellt"


================================================================
SCHRITT 2 — Code von GitHub laden
================================================================

cd /tmp && git clone -b claude/explore-new-features-LZPos https://github.com/Testerking434/rust-maps.git selfcore-temp


================================================================
SCHRITT 3 — Dateien kopieren
================================================================

cp /tmp/selfcore-temp/SELFCORE/SERVER/server.js /var/www/genselfcore/server.js
cp /tmp/selfcore-temp/SELFCORE/SERVER/package.json /var/www/genselfcore/package.json
mkdir -p /var/www/genselfcore/data/lessons /var/www/genselfcore/db
cp /tmp/selfcore-temp/SELFCORE/SERVER/data/*.json /var/www/genselfcore/data/
cp /tmp/selfcore-temp/SELFCORE/SERVER/data/lessons/*.json /var/www/genselfcore/data/lessons/
rm -rf /tmp/selfcore-temp

echo "✅ Dateien kopiert"


================================================================
SCHRITT 4 — Dependencies installieren
================================================================

cd /var/www/genselfcore && npm install

echo "✅ Installiert"


================================================================
SCHRITT 5 — Server neu starten
================================================================

pm2 restart genselfcore

echo "✅ Server gestartet"


================================================================
SCHRITT 6 — Testen
================================================================

curl https://api.genselfcore.de/health
curl https://api.genselfcore.de/v1/abo/tiers
curl https://api.genselfcore.de/v1/dna-test/questions


================================================================
FERTIG!
================================================================

Login-System:
  1. Sign in with Apple  (Haupt-Login, ein Tap)
  2. Sign in with Google (zweite Option)
  3. E-Mail              (für die die es wollen)

Kein SMTP nötig. Kein Passwort-Reset nötig.
Apple + Google machen alles.

Die .env ist simpel — nur PORT + JWT_SECRET.

Später optional hinzufügen:
  R2_ACCOUNT_ID=...    (für Audio-Streaming)
  R2_ACCESS_KEY=...
  R2_SECRET_KEY=...
  APPLE_SHARED_SECRET=... (für In-App-Kauf Validierung)

================================================================
PM2 BEFEHLE
================================================================

pm2 status                   — Status
pm2 logs genselfcore         — Logs
pm2 restart genselfcore      — Neustart
pm2 stop genselfcore         — Stoppen

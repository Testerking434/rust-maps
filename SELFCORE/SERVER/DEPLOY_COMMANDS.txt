'use strict';
require('dotenv').config();
const express   = require('express');
const bcrypt    = require('bcryptjs');
const jwt       = require('jsonwebtoken');
const cors      = require('cors');
const path      = require('path');
const fs        = require('fs');

const app        = express();
const PORT       = process.env.PORT || 8001;
const JWT_SECRET = process.env.JWT_SECRET || 'changeme';

app.use(cors());
app.use(express.json());

const users             = {};
const userById          = {};
const referrals         = {};
const codeToUser        = {};
const signalExtensions  = {};
const processedReferrals = new Set();
const userReferredBy    = {};
const rewardIDs         = new Set();
const registrationLocks = new Set();
const idempotencyCache  = {};

function generateReferralCode(name) {
  const base = ((name || '').substring(0,3).toUpperCase().replace(/[^A-Z]/g,'X')) || 'REF';
  const suffix = Math.random().toString(36).substring(2,5).toUpperCase();
  const code = base + suffix;
  return codeToUser[code] ? generateReferralCode(name) : code;
}

function createRewardId(prefix) {
  let id;
  do { id = (prefix||'r') + '_' + Date.now() + '_' + Math.random().toString(36).substring(2,6); }
  while (rewardIDs.has(id));
  rewardIDs.add(id);
  return id;
}

function auth(req, res, next) {
  const h = req.headers.authorization;
  if (!h || !h.startsWith('Bearer ')) return res.status(401).json({error:'Nicht autorisiert.'});
  try { req.user = jwt.verify(h.split(' ')[1], JWT_SECRET); next(); }
  catch { return res.status(401).json({error:'Token ungültig.'}); }
}

function idempotency(req, res, next) {
  const key = req.headers['idempotency-key'];
  if (!key) return next();
  const now = Date.now();
  Object.keys(idempotencyCache).forEach(k => { if (idempotencyCache[k].exp < now) delete idempotencyCache[k]; });
  if (idempotencyCache[key]) return res.json(idempotencyCache[key].data);
  res.cache = d => { idempotencyCache[key] = {data:d, exp:now+1800000}; };
  next();
}

function referralResponse(userId) {
  const d = referrals[userId];
  if (!d) return {referralCode:'',referralLink:'',totalReferred:0,pendingReferred:0,earnedDays:0,badges:[],rewards:[],referredFriends:[]};
  return {referralCode:d.code,referralLink:`https://genselfcore.de/join?ref=${d.code}`,totalReferred:d.totalReferred,pendingReferred:0,earnedDays:d.earnedDays,badges:d.badges,rewards:d.rewards,referredFriends:d.friends};
}

function processReferral(newUserId, code, name) {
  const pair = `${newUserId}:${code}`;
  if (processedReferrals.has(pair)) return {skip:true};
  const referrerId = codeToUser[code];
  if (!referrerId || referrerId === newUserId || userReferredBy[newUserId]) return {skip:true};
  processedReferrals.add(pair);
  userReferredBy[newUserId] = code;
  if (!referrals[referrerId]) referrals[referrerId] = {code,totalReferred:0,earnedDays:0,badges:[],rewards:[],friends:[],badgeTitles:new Set()};
  const ref = referrals[referrerId];
  ref.totalReferred++;
  ref.earnedDays += 3;
  signalExtensions[referrerId] = (signalExtensions[referrerId]||0) + 3;
  const colors = ['#F5A623','#FF6B9D','#4A90D9','#FF6B35','#5CB85C','#00C9C9'];
  ref.friends.push({id:newUserId,firstName:(name||'Freund').split(' ')[0],joinedAt:new Date().toISOString(),isActive:true,avatarColor:colors[Math.floor(Math.random()*colors.length)]});
  ref.rewards.push({id:createRewardId('rw'),type:'signal_days',description:`${name||'Dein Freund'} hat sich registriert! +3 Tage`,daysGranted:3,earnedAt:new Date().toISOString(),isNew:true});
  const milestones = [
    {required:1, title:'CONNECTOR',emoji:'🤝',color:'#5CB85C',extraDays:0,track:null},
    {required:3, title:'CATALYST', emoji:'⚡️',color:'#00C9C9',extraDays:0,track:'gratitude-flow'},
    {required:5, title:'PIONEER',  emoji:'🔥',color:'#FF6B35',extraDays:7,track:null},
    {required:10,title:'LEGEND',   emoji:'👑',color:'#F5A623',extraDays:30,track:null},
  ];
  const ms = milestones.find(m => m.required === ref.totalReferred);
  if (ms && !ref.badgeTitles.has(ms.title)) {
    ref.badgeTitles.add(ms.title);
    ref.badges.push({id:createRewardId('b'),title:ms.title,emoji:ms.emoji,earnedAt:new Date().toISOString(),colorHex:ms.color});
    if (ms.extraDays > 0) { signalExtensions[referrerId]=(signalExtensions[referrerId]||0)+ms.extraDays; ref.earnedDays+=ms.extraDays; ref.rewards.push({id:createRewardId('ms'),type:'signal_days',description:`${ms.title}! +${ms.extraDays} Tage`,daysGranted:ms.extraDays,earnedAt:new Date().toISOString(),isNew:true}); }
    if (ms.track) ref.rewards.push({id:createRewardId('tr'),type:'exclusive_track',description:'Gratitude Flow freigeschaltet',daysGranted:0,earnedAt:new Date().toISOString(),isNew:true});
  }
  console.log(`Referral: ${name} via ${code} → Referrer ${referrerId} hat ${ref.totalReferred} Freunde`);
  return {success:true};
}

app.get('/health', (_,res) => res.json({status:'ok',app:'GEN:SELFCORE',port:PORT}));

app.post('/v1/auth/register', idempotency, async (req,res) => {
  const {name,email,password,referralCode} = req.body;
  if (!email||!password||!name) return res.status(400).json({error:'Alle Felder erforderlich.'});
  const norm = email.toLowerCase().trim();
  if (registrationLocks.has(norm)) return res.status(429).json({error:'Bitte warten.'});
  registrationLocks.add(norm);
  try {
    if (users[norm]) return res.status(409).json({error:'E-Mail bereits registriert.'});
    const hash = await bcrypt.hash(password, 12);
    const id   = 'u_'+Date.now()+'_'+Math.random().toString(36).substring(2,7);
    const tok  = jwt.sign({userId:id,email:norm},JWT_SECRET,{expiresIn:'90d'});
    const u    = {id,name:name.trim(),email:norm,passwordHash:hash,selfcoreType:'ACHIEVER',dimensions:{selbstkenntnis:5,authentizitaet:5,klarheit:5,mut:5,verbindung:5}};
    users[norm]=u; userById[id]=u;
    let freeDays=0;
    if (referralCode) { const r=processReferral(id,referralCode.toUpperCase().trim(),name.trim()); if(r.success){signalExtensions[id]=(signalExtensions[id]||0)+3;freeDays=3;} }
    const out={token:tok,user:{id,name:u.name,email:norm,selfcoreType:u.selfcoreType,dimensions:u.dimensions},signalFreeDays:freeDays};
    if(res.cache)res.cache(out);
    return res.status(201).json(out);
  } finally { registrationLocks.delete(norm); }
});

app.post('/v1/auth/login', async (req,res) => {
  const {email,password} = req.body;
  if (!email||!password) return res.status(400).json({error:'E-Mail und Passwort erforderlich.'});
  const u = users[email.toLowerCase().trim()];
  if (!u) return res.status(401).json({error:'E-Mail oder Passwort falsch.'});
  if (!await bcrypt.compare(password,u.passwordHash)) return res.status(401).json({error:'E-Mail oder Passwort falsch.'});
  const tok = jwt.sign({userId:u.id,email:u.email},JWT_SECRET,{expiresIn:'90d'});
  return res.json({token:tok,user:{id:u.id,name:u.name,email:u.email,selfcoreType:u.selfcoreType,dimensions:u.dimensions}});
});

app.post('/v1/auth/reset-password', (_,res) => res.json({message:'Falls registriert, wurde ein Link gesendet.'}));
app.post('/v1/auth/logout', auth, (_,res) => res.json({message:'Abgemeldet.'}));

app.get('/v1/user/profile', auth, (req,res) => {
  const u=userById[req.user.userId];
  if(!u)return res.status(404).json({error:'Nicht gefunden.'});
  return res.json({id:u.id,name:u.name,email:u.email,selfcoreType:u.selfcoreType,dimensions:u.dimensions});
});

app.get('/v1/user/courses', auth, (_,res) => {
  try { const p=path.join(__dirname,'data','courses.json'); if(fs.existsSync(p))return res.json(JSON.parse(fs.readFileSync(p,'utf8'))); } catch{}
  return res.json([{id:'awakening',title:'AWAKENING',subtitle:'Erwache zu dir selbst',lessonCount:42,coverImageURL:''},{id:'origin',title:'ORIGIN',subtitle:'Dein DNA-Profil',lessonCount:10,coverImageURL:''}]);
});

app.get('/v1/user/signal-status', auth, (req,res) => {
  const d=signalExtensions[req.user.userId]||0;
  return res.json({isSubscribed:false,freeDaysRemaining:d,hasAccess:d>0});
});

app.get('/v1/user/weekly-stats', auth, (_,res) => res.json({checkInsCount:0,actionsCompleted:0,dominantMood:null,dimensionChanges:{}}));

const checkIns={};
app.post('/v1/checkins', auth, (req,res) => {
  const id=req.user.userId;
  if(!checkIns[id])checkIns[id]=[];
  checkIns[id].push({...req.body,savedAt:new Date().toISOString()});
  return res.json({message:'Gespeichert.'});
});

app.get('/v1/courses/:courseId/lessons', auth, (req,res) => {
  try { const p=path.join(__dirname,'data','lessons',`${req.params.courseId}.json`); if(fs.existsSync(p))return res.json(JSON.parse(fs.readFileSync(p,'utf8'))); } catch{}
  return res.json([]);
});

app.get('/v1/courses/:courseId/lessons/:lessonId', auth, (req,res) => {
  try { const p=path.join(__dirname,'data','lessons',req.params.courseId,`${req.params.lessonId}.json`); if(fs.existsSync(p))return res.json(JSON.parse(fs.readFileSync(p,'utf8'))); } catch{}
  return res.status(404).json({error:'Nicht gefunden.'});
});

app.get('/v1/lessons/:lessonId/workbook-url', auth, (_,res) => res.json({url:''}));
app.post('/v1/lessons/:lessonId/complete', auth, (_,res) => res.json({message:'Abgeschlossen.'}));
app.get('/v1/signal/tracks', auth, (_,res) => { try{const p=path.join(__dirname,'data','signal_tracks.json');if(fs.existsSync(p))return res.json(JSON.parse(fs.readFileSync(p,'utf8')));}catch{} return res.json([]); });
app.get('/v1/signal/tracks/:id/url', auth, (_,res) => res.json({url:''}));

app.get('/v1/referral/code', auth, (req,res) => {
  const id=req.user.userId, u=userById[id];
  if(!referrals[id]){const code=generateReferralCode(u?.name||'');referrals[id]={code,totalReferred:0,earnedDays:0,badges:[],rewards:[],friends:[],badgeTitles:new Set()};codeToUser[code]=id;}
  return res.json(referralResponse(id));
});

app.post('/v1/referral/code', idempotency, auth, (req,res) => {
  const id=req.user.userId, u=userById[id];
  if(!referrals[id]){const code=generateReferralCode(u?.name||'');referrals[id]={code,totalReferred:0,earnedDays:0,badges:[],rewards:[],friends:[],badgeTitles:new Set()};codeToUser[code]=id;}
  const out=referralResponse(id);
  if(res.cache)res.cache(out);
  return res.json(out);
});

app.get('/v1/referral/data', auth, (req,res) => res.json(referralResponse(req.user.userId)));

app.post('/v1/referral/rewards/seen', idempotency, auth, (req,res) => {
  const d=referrals[req.user.userId];
  if(d)d.rewards=d.rewards.map(r=>({...r,isNew:false}));
  const out={};
  if(res.cache)res.cache(out);
  return res.json(out);
});

['data','data/lessons'].forEach(d => { const p=path.join(__dirname,d); if(!fs.existsSync(p))fs.mkdirSync(p,{recursive:true}); });

app.listen(PORT, () => console.log(`GEN:SELFCORE läuft auf Port ${PORT}`));

SCHRITT 11 — Nginx Config erstellen
----------------------------------------------------------------

cat > /etc/nginx/sites-available/genselfcore << 'EOF'
server {
    listen 80;
    server_name api.genselfcore.de;
    location / {
        proxy_pass         http://localhost:8001;
        proxy_http_version 1.1;
        proxy_set_header   Host $host;
        proxy_set_header   X-Real-IP $remote_addr;
        proxy_set_header   X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
server {
    listen 80;
    server_name genselfcore.de www.genselfcore.de;
    root /var/www/genselfcore;
    index index.html;
    location /join {
        try_files $uri /join.html;
    }
    location /.well-known/ {
        alias /var/www/genselfcore/.well-known/;
        default_type application/json;
    }
    location / {
        try_files $uri $uri/ /index.html;
    }
}
EOF
